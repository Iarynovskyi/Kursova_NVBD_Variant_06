   Запуск: SSMS → Connect → Analysis Services → DESKTOP-I5COGTD → база Polyclinic_OLAP
           → New Query (MDX); виділити один запит і виконати (F5).
   Обчислювані міри, набори та KPI визначені в MDX-скрипті куба
   (вкладка Calculations у Visual Studio).

-- 1. Time Intelligence: візити по роках, YTD, попередній рік (PY), приріст YoY %
SELECT
    { [Measures].[Кількість візитів], [Measures].[Візити попереднього року (PY)], [Measures].[Приріст візитів YoY],
      [Measures].[Вартість консультацій у грн], [Measures].[Приріст вартості консультацій YoY] } ON COLUMNS,
    { [Дата].[Календар].[Рік].&[2016] : [Дата].[Календар].[Рік].&[2025] } ON ROWS
FROM [Polyclinic DW];

-- 2. Year To Date по місяцях 2024 року та ковзне середнє за 3 місяці
SELECT
    { [Measures].[Кількість візитів], [Measures].[Візити з початку року (YTD)], [Measures].[Ковзне середнє візитів (3 періоди)] } ON COLUMNS,
    DESCENDANTS([Дата].[Календар].[Рік].&[2024], [Дата].[Календар].[Місяць]) ON ROWS
FROM [Polyclinic DW];

-- 3. Ranking / Top-N: десять найбільш завантажених лікарів за 2024 рік, позиція в рейтингу і частка
SELECT
    { [Measures].[Кількість візитів], [Measures].[Рейтинг лікаря за візитами], [Measures].[Частка лікаря у візитах],
      [Measures].[Середня тривалість прийому у хв] } ON COLUMNS,
    [Топ-10 лікарів за візитами] ON ROWS
FROM [Polyclinic DW]
WHERE ( [Дата].[Календар].[Рік].&[2024] );

-- 4. Бізнес-метрики по відділеннях: дохід на годину прийому, частка безкоштовних візитів НСЗУ, частка одужань
SELECT
    { [Measures].[Кількість візитів], [Measures].[Дохід на годину прийому у грн], [Measures].[Частка безкоштовних візитів НСЗУ],
      [Measures].[Частка одужань], [Measures].[Середня тривалість лікування у днях] } ON COLUMNS,
    NON EMPTY [Лікар].[Структура поліклініки].[Відділення].MEMBERS ON ROWS
FROM [Polyclinic DW]
WHERE ( [Дата].[Календар].[Рік].&[2025] );

-- 5. KPI: ефективність лікування та дохід від консультацій по роках (значення, ціль, статус, тренд)
SELECT
    { KPIValue("Ефективність лікування"), KPIGoal("Ефективність лікування"), KPIStatus("Ефективність лікування"), KPITrend("Ефективність лікування"),
      KPIValue("Дохід від консультацій"), KPIGoal("Дохід від консультацій"), KPIStatus("Дохід від консультацій") } ON COLUMNS,
    { [Дата].[Календар].[Рік].&[2017] : [Дата].[Календар].[Рік].&[2025] } ON ROWS
FROM [Polyclinic DW];

-- 6. Статистика захворювань (звіт варіанту №5): розділи МКХ-10 × вікові групи, обчислюваний член «Дорослі від 18 років»
SELECT
    { [Вікова група].[Вікова група].[Вікова група].MEMBERS, [Вікова група].[Вікова група].[Усі].[Дорослі від 18 років] } ON COLUMNS,
    NON EMPTY [Хвороба].[МКХ-10].[Розділ].MEMBERS ON ROWS
FROM [Polyclinic DW]
WHERE ( [Measures].[Кількість діагнозів], [Дата].[Календар].[Рік].&[2025] );

-- 7. Завантаженість лікарів по місяцях (звіт варіанту №3): ROLAP-група мір, гранулярність — місяць
SELECT
    { [Measures].[Візитів за місяць], [Measures].[Робочих днів], [Measures].[Візитів на робочий день],
      [Measures].[Мінімальна тривалість прийому у хв], [Measures].[Максимальна тривалість прийому у хв] } ON COLUMNS,
    DESCENDANTS([Дата].[Календар].[Рік].&[2025], [Дата].[Календар].[Місяць]) ON ROWS
FROM [Polyclinic DW]
WHERE ( [Лікар].[Структура поліклініки].[Відділення].&[Кардіологічне відділення] );

-- 8. Ефективність лікування (звіт варіанту №4): тяжкість × результат, частка одужань і середній лікарняний
SELECT
    { [Measures].[Кількість діагнозів], [Measures].[Частка одужань], [Measures].[Лікарняних на діагноз у днях] } ON COLUMNS,
    NON EMPTY [Клінічний результат].[Тяжкість].[Тяжкість].MEMBERS ON ROWS
FROM [Polyclinic DW];

-- 9. Перспектива «Клінічна аналітика»: SCD2 — візити за версіями пацієнтів району; дата результату (рольовий вимір)
SELECT
    { [Measures].[Кількість діагнозів], [Measures].[Унікальних пацієнтів] } ON COLUMNS,
    NON EMPTY [Дата результату].[Календар].[Рік].MEMBERS ON ROWS
FROM [Клінічна аналітика]
WHERE ( [Джерело даних].[Джерело даних].&[OLTP] );

-- 10. Фармація: топ-10 препаратів за вартістю, частка державної компенсації
SELECT
    { [Measures].[Вартість ліків у грн], [Measures].[Кількість рецептів], [Measures].[Частка компенсації ліків] } ON COLUMNS,
    [Топ-10 препаратів за вартістю] ON ROWS
FROM [Фармація]
WHERE ( [Дата].[Календар].[Рік].&[2025] );
